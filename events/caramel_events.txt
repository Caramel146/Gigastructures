namespace = caramel

#pls AI be more relevant in endgame
country_event = {
    id = caramel.1
    hide_window = yes

    is_triggered_only = yes

    trigger = {
        is_ai = yes
        is_country_type = default
    }

    immediate = {
        add_modifier = {
            modifier = caramel_ai_scaling
            years = -1
            mult = trigger:years_passed
        }
        if = {
            limit = {
                mid_game_years_passed > 0
            }
            add_modifier = {
                modifier = caramel_ai_midgame_scaling
                years = -1
                mult = trigger:mid_game_years_passed
            }
        }
        if = {
            limit = {
                end_game_years_passed > 0
            }
            add_modifier = {
                modifier = caramel_ai_endgame_scaling
                years = -1
                mult = trigger:end_game_years_passed
            }
        }
    }
}
