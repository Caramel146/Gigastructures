    }
}

#Roll a new random event from the list every x months into the situation
situation_event = {
    id = giga_scorched.003
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        change_variable = {
            which = scorched_situation_timer
            value = 1
        }

        if = {
            limit = {
                check_variable = {
                    which = scorched_situation_timer
                    value >= 24
                }
            }
            export_trigger_value_to_variable = {
                trigger = situation_progress
                variable = scorched_situation_final_position
            }
            owner = {
                set_variable = {
                    which = scorched_situation_final_position
                    value = prev.scorched_situation_final_position
                }
                random_list = {
                    1 = {
                        country_event = { id = giga_scorched.004 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_1 #checks triggered instead of concluded in case it rolls again in the time someone doesn't answer
                        }
                    }
                    1 = {
                        country_event = { id = giga_scorched.004 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_2
                        }
                    }
                    1 = {
                        country_event = { id = giga_scorched.004 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_3
                        }
                    }
                    1 = {
                        country_event = { id = giga_scorched.004 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_4
                        }
                    }
                }
            }
            set_variable = {
                which = scorched_situation_timer
                value = 0
            }
        }
    }
}
