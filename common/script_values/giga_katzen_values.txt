katzen_systems = {
    complex_trigger_modifier = {
        trigger = count_system_within_border
        mode = add
    }
}

katzen_or_vassal_systems = {
    base = 0
    complex_trigger_modifier = {
        trigger = count_system
        mode = add
        parameters = {
            limit = {
                owner = {
                    OR = {
                        is_country_type = katzenartig_imperium
                        katzen_vassal = yes
                    }
                }
            }
        }
    }
}

katzen_reformation_collapse_valid_splinter_systems = {
    base = 0
    complex_trigger_modifier = {
        trigger = count_system_within_border
        mode = add
        parameters = {
            limit = {
                has_katzen_colony = yes #must have at least one colony to be a splinter capital
                any_system = { #must be a sufficient distance away from Jublio
                    distance = {
                        source = prev
                        type = euclidean
                        min_distance >= 1
                        max_distance <= 10
                    }
                    is_same_value = event_target:giga_jublio_system
                }
            }
        }
    }
}

katzen_or_vassal_systems_percent = {
    base = 0
    add = value:katzen_or_vassal_systems
    divide = trigger:num_galaxy_systems
}

kaiser_size = {
    base = 0

    add = value:katzen_or_vassal_systems_percent

        trigger = count_owned_pop_amount

    complex_trigger_modifier = {

    }
